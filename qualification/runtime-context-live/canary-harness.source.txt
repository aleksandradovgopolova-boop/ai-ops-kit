    with tempfile.TemporaryDirectory(prefix='aiops-native-qualification-') as temp:
        base=Path(temp);control=base/'control';workspace=base/'workspace';home=base/'codex-home'

               for event in ('SessionStart','PreToolUse','PostToolUse')}

        if runtime=='codex':
            auth=Path('/Users/sasad/.codex/auth.json')

            env['CODEX_HOME']=str(home)
            (home/'hooks.json').write_text(json.dumps({'hooks':hooks}))
            cmd=['/Users/sasad/.local/bin/codex','exec','--ignore-user-config','--ephemeral',
                 '--dangerously-bypass-hook-trust','--skip-git-repo-check','--sandbox','workspace-write',
                 '--json','--model','gpt-5.6-terra','-c','model_reasoning_effort="low"',
                 '-c','shell_environment_policy.inherit="none"','--output-last-message',str(control/'answer.txt'),PROMPT]

            (control/'settings.json').write_text(json.dumps({'hooks':hooks}))

            except ValueError:pass
